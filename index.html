Automação SAP - Modo Detecção Global Permanente

Esta versão resolve o problema de elementos destruídos pelo SAP monitorando a página de forma global.

(function() {
    let temporizador;
    let executandoClique = false;

    // Defina aqui o ID do botão de confirmação do seu pop-up
    const SELETOR_BOTAO_CONFIRMACAO = "[id$='COLOQUE_AQUI_O_ID_DO_BOTAO_DE_CONFIRMACAO']";

    console.log("🚀 Automação SAP Global Ativada! Monitorando bipes continuamente...");

    function dispararCliqueReal(elemento) {
        elemento.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, cancelable: true }));
        elemento.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, cancelable: true }));
        elemento.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true }));
    }

    function esperarBotaoConfirmacao() {
        console.log("👀 Aguardando o pop-up de confirmação aparecer...");
        
        const observadorPopup = new MutationObserver(function(mutations, observer) {
            const botaoConfirmar = document.querySelector(SELETOR_BOTAO_CONFIRMACAO);
            
            if (botaoConfirmar) {
                console.log("✅ Botão de confirmação detectado! Clicando...");
                dispararCliqueReal(botaoConfirmar);
                observer.disconnect(); 
                
                // Libera o script para aceitar novos bipes após fechar o pop-up
                setTimeout(() => { 
                    executandoClique = false; 
                    console.log("🔄 Pronto para o próximo lote!");
                }, 1000);
            }
        });

        observadorPopup.observe(document.body, { childList: true, subtree: true });
    }

    // OUVINTE GLOBAL: Captura qualquer digitação na página inteira, independente do elemento sumir ou reaparecer
    document.body.addEventListener("input", function(evento) {
        const campo = evento.target;

        // Verifica se o campo que disparou o evento é o nosso segundo input de lote
        if (campo && campo.matches("[id$='LoteInput2-inner']")) {
            
            // Bloqueia bipes duplicados enquanto o clique anterior está processando
            if (executandoClique) return;

            clearTimeout(temporizador); 

            if (campo.value.trim() !== "") {
                console.log("✍️ Bipe detectado no segundo lote. Aguardando 3 segundos...");

                temporizador = setTimeout(function() {
                    executandoClique = true; // Trava para evitar cliques múltiplos

                    // Avisa o framework SAP que o campo mudou
                    campo.dispatchEvent(new Event('change', { bubbles: true }));
                    campo.dispatchEvent(new Event('liveChange', { bubbles: true }));

                    const botaoProximo = document.querySelector("[id$='nextButton-BDI-content']");
                    
                    if (botaoProximo) {
                        console.log("⏱️ 3 segundos atingidos! Apertando o botão Próximo...");
                        esperarBotaoConfirmacao();
                        dispararCliqueReal(botaoProximo);
                    } else {
                        console.error("❌ Erro: O botão 'Próximo' não foi localizado.");
                        executandoClique = false; // Destrava se houver erro
                    }
                }, 3000); 
            }
        }
    });
})();
