Automação SAP - Versão Final Concluída

O código abaixo já está configurado com todos os IDs corretos. Clique no botão para copiar e colar no Console (F12).

(function() {
    let temporizadorClique;
    let executandoFluxo = false;
    let ultimoConteudo = "";

    // ID do seu botão de confirmação configurado com sucesso!
    const SELETOR_BOTAO_CONFIRMACAO = "[id$='_IDGenButton2-BDI-content']";

    console.log("🚀 Rastreador Ativo SAP Iniciado! Monitorando em segundo plano...");

    function dispararCliqueReal(elemento) {
        elemento.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, cancelable: true }));
        elemento.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, cancelable: true }));
        elemento.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true }));
    }

    function fecharPopupConfirmacao() {
        const botaoConfirmar = document.querySelector(SELETOR_BOTAO_CONFIRMACAO);
        if (botaoConfirmar) {
            console.log("✅ Botão de confirmação detectado! Clicando...");
            dispararCliqueReal(botaoConfirmar);
            
            // Aguarda o SAP processar o fechamento e libera o robô para o próximo lote
            setTimeout(() => {
                ultimoConteudo = "";
                executandoFluxo = false;
                console.log("🔄 Sistema resetado. Pronto para o próximo ciclo!");
            }, 1500);
        } else {
            // Se o pop-up ainda não apareceu na tela, tenta de novo em 300ms
            setTimeout(fecharPopupConfirmacao, 300);
        }
    }

    // RASTREADOR PRINCIPAL: Roda a cada 500 milissegundos por fora dos eventos do SAP
    setInterval(function() {
        // Se já estivermos no meio de um clique/processamento, ignora a checagem temporariamente
        if (executandoFluxo) return;

        const campo = document.querySelector("[id$='LoteInput2-inner']");
        
        if (campo) {
            const valorAtual = campo.value.trim();

            // Detecta se algo novo foi digitado ou bipado no campo
            if (valorAtual !== "" && valorAtual !== ultimoConteudo) {
                ultimoConteudo = valorAtual; // Atualiza o último texto registrado
                console.log("✍️ Texto detectado no segundo lote pelo rastreador. Aguardando pausa de 3 segundos...");

                // Reseta qualquer contagem anterior para garantir os 3 segundos após o término do bipe
                clearTimeout(temporizadorClique);

                temporizadorClique = setTimeout(function() {
                    executandoFluxo = true; // Trava o rastreador para evitar cliques duplos

                    console.log("⏱️ 3 segundos atingidos! Notificando o framework SAP...");
                    campo.dispatchEvent(new Event('change', { bubbles: true }));
                    campo.dispatchEvent(new Event('liveChange', { bubbles: true }));

                    const botaoProximo = document.querySelector("[id$='nextButton-BDI-content']");
                    
                    if (botaoProximo) {
                        console.log("👉 Apertando o botão Próximo...");
                        dispararCliqueReal(botaoProximo);
                        
                        // Inicia a busca ativa pelo botão de confirmação que vai surgir
                        fecharPopupConfirmacao();
                    } else {
                        console.error("❌ Erro: O botão 'Próximo' não estava visível ou pronto.");
                        executandoFluxo = false; // Destrava em caso de falha externa
                    }
                }, 3000);
            } else if (valorAtual === "") {
                // Se o campo for limpo pelo sistema, reseta o histórico do conteúdo
                ultimoConteudo = "";
            }
        }
    }, 500);
})();
