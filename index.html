Automação SAP - Campo de Faixa Desbloqueado

Esta versão usa um bloco de texto editável imune aos bloqueios de estilo do SAP. Copie e cole no Console (F12).

(function() {
    let temporizadorClique;
    let executandoFluxo = false;
    let ultimoConteudo = "";

    // IDs configurados do seu sistema SAP
    const SELETOR_BOTAO_CONFIRMACAO = "[id$='_IDGenButton2-BDI-content']";
    const SELETOR_CAMPO_FAIXA = "[id$='FaixaInput-inner'], [id$='faixaInput-inner'], input[id*='Faixa']"; 

    console.log("🚀 Rastreador Ativo SAP Iniciado (Campo Desbloqueado)!");

    // 1. CRIA O PAINEL VISUAL FLUTUANTE COM ÁREA EDITÁVEL (IMUNE AO BLOQUEIO DO SAP)
    const painel = document.createElement("div");
    painel.id = "painel-automacao-sap";
    painel.style.cssText = "position: fixed !important; top: 15px !important; left: 15px !important; z-index: 999999 !important; background-color: #1e272e !important; color: #ffffff !important; padding: 15px !important; border-radius: 8px !important; box-shadow: 0 6px 20px rgba(0,0,0,0.5) !important; font-family: Arial, sans-serif !important; font-size: 14px !important; border: 2px solid #ff9f43 !important; width: 160px !important; text-align: left !important;";
    
    // Usamos contenteditable="true" em uma div comum para o SAP não conseguir ocultar o campo
    painel.innerHTML = `
        
🤖 Robô SAP

        
        

    `;
    document.body.appendChild(painel);

    function dispararCliqueReal(elemento) {
        elemento.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, cancelable: true }));
        elemento.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, cancelable: true }));
        elemento.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true }));
    }

    function fecharPopupConfirmacao() {
        const botaoConfirmar = document.querySelector(SELETOR_BOTAO_CONFIRMACAO);
        if (botaoConfirmar) {
            console.log("✅ Botão de confirmação detectado! Clicando...");
            dispararCliqueReal(botaoConfirmar);
            
            setTimeout(() => {
                ultimoConteudo = "";
                executandoFluxo = false;
                console.log("🔄 Sistema resetado. Pronto para o próximo ciclo!");
            }, 1500);
        } else {
            setTimeout(fecharPopupConfirmacao, 300);
        }
    }

    // RASTREADOR PRINCIPAL (Roda a cada 500ms)
    setInterval(function() {
        // 2. INJEÇÃO DA FAIXA EM TEMPO REAL
        const campoFaixaSap = document.querySelector(SELETOR_CAMPO_FAIXA);
        
        // Captura o texto de dentro da div editável usando innerText
        const divFaixa = document.getElementById("campo-faixa-fixa");
        const valorFaixaDefinida = divFaixa ? divFaixa.innerText.trim() : "";

        if (campoFaixaSap && valorFaixaDefinida !== "") {
            if (campoFaixaSap.value !== valorFaixaDefinida) {
                campoFaixaSap.value = valorFaixaDefinida;
                campoFaixaSap.dispatchEvent(new Event('change', { bubbles: true }));
                campoFaixaSap.dispatchEvent(new Event('liveChange', { bubbles: true }));
                console.log("✏️ Faixa corrigida automaticamente para: " + valorFaixaDefinida);
            }
        }

        // 3. FLUXO DE AVANÇAR DO SEGUNDO LOTE
        if (executandoFluxo) return;

        const campoLote = document.querySelector("[id$='LoteInput2-inner']");
        
        if (campoLote) {
            const valorAtual = campoLote.value.trim();

            if (valorAtual !== "" && valorAtual !== ultimoConteudo) {
                ultimoConteudo = valorAtual;
                console.log("✍️ Texto detectado no segundo lote. Aguardando 3 segundos...");

                clearTimeout(temporizadorClique);

                temporizadorClique = setTimeout(function() {
                    executandoFluxo = true; 

                    console.log("⏱️ 3 segundos atingidos! Confirmando dados no SAP...");
                    campoLote.dispatchEvent(new Event('change', { bubbles: true }));
                    campoLote.dispatchEvent(new Event('liveChange', { bubbles: true }));

                    const botaoProximo = document.querySelector("[id$='nextButton-BDI-content']");
                    
                    if (botaoProximo) {
                        console.log("👉 Apertando o botão Próximo...");
                        dispararCliqueReal(botaoProximo);
                        fecharPopupConfirmacao();
                    } else {
                        console.error("❌ Erro: O botão 'Próximo' não estava visível.");
                        executandoFluxo = false;
                    }
                }, 3000);
            } else if (valorAtual === "") {
                ultimoConteudo = "";
            }
        }
    }, 500);
})();
