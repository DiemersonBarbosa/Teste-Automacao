Código de Automação SAP (Versão Blindada v2)

Esta versão corrige falhas de sincronização e IDs dinâmicos do SAP. Copie e cole no Console (F12).

(function() {
    let temporizador;

    // Busca inteligente: procura elementos cujo ID TERMINE com o termo correto (evita problemas se o prefixo mudar)
    const campo = document.querySelector("[id$='LoteInput2-inner']");

    if (!campo) {
        console.error("⚠️ Erro: Campo de lote não encontrado. Verifique se a tela está totalmente carregada.");
        return;
    }

    console.log("🚀 Automação SAP Blindada Ativada! Monitorando campo...");

    campo.addEventListener("input", function() {
        clearTimeout(temporizador); 

        if (campo.value.trim() !== "") {
            console.log("✍️ Detectada digitação/bipe. Aguardando 3 segundos...");

            temporizador = setTimeout(function() {
                // Força o SAP a reconhecer que o texto foi inserido antes de clicar
                campo.dispatchEvent(new Event('change', { bubbles: true }));
                campo.dispatchEvent(new Event('liveChange', { bubbles: true }));

                // Busca inteligente para o botão (procura por ID que termina com o termo do botão)
                const botao = document.querySelector("[id$='nextButton-BDI-content']");
                
                if (botao) {
                    console.log("⏱️ 3 segundos atingidos! Disparando clique simulado...");
                    
                    // Simulação profunda de clique físico exigida pelo SAP UI5
                    botao.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, cancelable: true }));
                    botao.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, cancelable: true }));
                    botao.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true }));
                } else {
                    console.error("❌ Erro: O botão 'Próximo' não foi localizado na tela.");
                }
            }, 3000); 
        }
    });
})();
