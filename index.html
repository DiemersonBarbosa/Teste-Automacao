Automação SAP - Fluxo Completo com Confirmação

Esta versão clica no botão "Próximo", aguarda o pop-up de confirmação aparecer na tela e clica nele automaticamente.

⚠️ Atenção: No código abaixo, substitua o texto COLOQUE_AQUI_O_ID_DO_BOTAO_DE_CONFIRMACAO pelo ID real ou pelo texto do botão de confirmação que aparece no pop-up (ex: "Sim", "OK" ou o ID dele obtido pelo F12). 

(function() {
    let temporizador;
    let inputEscutado = false;

    // 1. Configure aqui a identificação do botão de confirmação (pode ser o ID ou o texto dele)
    // Se for usar o ID, mantenha o [id$='...'] como nos outros. Se for texto, use o seletor abaixo.
    const SELETOR_BOTAO_CONFIRMACAO = "[id$='COLOQUE_AQUI_O_ID_DO_BOTAO_DE_CONFIRMACAO']";

    console.log("🚀 Automação SAP Avançada Iniciada! Monitorando o segundo campo...");

    // Função auxiliar para disparar o clique físico perfeito exigido pelo SAP
    function dispararCliqueReal(elemento) {
        elemento.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, cancelable: true }));
        elemento.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, cancelable: true }));
        elemento.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true }));
    }

    // Função que monitora a aparição do botão de confirmação no pop-up
    function esperarBotaoConfirmacao() {
        console.log("👀 Aguardando o pop-up de confirmação aparecer...");
        
        const observadorPopup = new MutationObserver(function(mutations, observer) {
            const botaoConfirmar = document.querySelector(SELETOR_BOTAO_CONFIRMACAO);
            
            if (botaoConfirmar) {
                console.log("✅ Botão de confirmação detectado! Clicando...");
                dispararCliqueReal(botaoConfirmar);
                observer.disconnect(); // Desliga este observador após clicar para poupar memória
            }
        });

        observadorPopup.observe(document.body, { childList: true, subtree: true });
    }

    // Função que aplica os ouvintes de evento assim que o segundo input aparece
    function iniciarMonitoramentoInput(campo) {
        if (inputEscutado) return;
        inputEscutado = true;
        console.log("🎯 Segundo campo de lote detectado na tela! Monitorando digitação...");

        campo.addEventListener("input", function() {
            clearTimeout(temporizador); 

            if (campo.value.trim() !== "") {
                console.log("✍️ Digitando/Bipando... Aguardando 3 segundos.");

                temporizador = setTimeout(function() {
                    campo.dispatchEvent(new Event('change', { bubbles: true }));
                    campo.dispatchEvent(new Event('liveChange', { bubbles: true }));

                    const botaoProximo = document.querySelector("[id$='nextButton-BDI-content']");
                    
                    if (botaoProximo) {
                        console.log("⏱️ 3 segundos atingidos! Apertando o botão Próximo...");
                        
                        // Liga a vigília pelo pop-up de confirmação ANTES de clicar no primeiro botão
                        esperarBotaoConfirmacao();
                        
                        // Clica no primeiro botão
                        dispararCliqueReal(botaoProximo);
                    } else {
                        console.error("❌ Erro: O botão 'Próximo' não foi localizado.");
                    }
                }, 3000); 
            }
        });
    }

    // Observador principal que fica olhando o surgimento do segundo campo de input
    const observadorPrincipal = new MutationObserver(function() {
        const segundoInput = document.querySelector("[id$='LoteInput2-inner']");
        if (segundoInput) {
            iniciarMonitoramentoInput(segundoInput);
        }
    });

    observadorPrincipal.observe(document.body, { childList: true, subtree: true });
})();
