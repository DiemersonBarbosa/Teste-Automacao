Código de Automação SAP

Clique no botão azul para copiar o script e cole no Console (F12) da página do SAP.

(function() {
    // IDs exatos do sistema SAP corrigidos com 'application'
    const idInputSap = "application-ZS_EWM_ENDEREC-manage-component---View1--LoteInput2-inner";
    const idBotaoSap = "application-ZS_EWM_ENDEREC-manage-component---View1--OrigemDestino-nextButton-BDI-content";
    
    let temporizador;
    const campo = document.getElementById(idInputSap);

    if (!campo) {
        console.error("⚠️ Erro: O input do lote não foi encontrado na tela. Certifique-se de estar na página correta.");
        return;
    }

    console.log("🚀 Automação SAP ativada! Monitorando o campo de lote...");

    // Escuta a digitação ou a leitura do bipe de código de barras
    campo.addEventListener("input", function() {
        clearTimeout(temporizador); // Reseta o tempo a cada novo caractere

        if (campo.value.trim() !== "") {
            console.log("✍️ Digitando/Bipando... Aguardando pausa de 3 segundos.");

            // Aguarda 3 segundos (3000ms) de silêncio antes de agir
            temporizador = setTimeout(function() {
                const botao = document.getElementById(idBotaoSap);
                
                if (botao) {
                    console.log("⏱️ 3 segundos atingidos! Acionando o botão próximo...");
                    
                    // Dispara um evento de clique real simulando o mouse físico para o SAPUI5
                    const eventoClique = new MouseEvent('click', { bubbles: true, cancelable: true });
                    botao.dispatchEvent(eventoClique);
                } else {
                    console.error("❌ Erro: O botão de avançar não foi localizado nesta tela.");
                }
            }, 3000); 
        }
    });
})();
