Código de Automação SAP (Suporte a Elementos Ocultos)

Esta versão monitora a tela para capturar o segundo input e o botão assim que eles surgirem na página. Copie e cole no Console (F12).

(function() {
    let temporizador;
    let inputEscutado = false;

    console.log("🚀 Automação Dinâmica SAP Iniciada! Monitorando o surgimento dos campos...");

    // Função que aplica os ouvintes de evento assim que o segundo input aparece
    function iniciarMonitoramentoInput(campo) {
        if (inputEscutado) return; // Evita duplicar o monitoramento
        inputEscutado = true;
        console.log("🎯 Segundo campo de lote detectado na tela! Monitorando digitação...");

        campo.addEventListener("input", function() {
            clearTimeout(temporizador); 

            if (campo.value.trim() !== "") {
                console.log("✍️ Digitando/Bipando no segundo lote... Aguardando 3 segundos.");

                temporizador = setTimeout(function() {
                    // Notifica o framework SAP UI5 que o campo mudou de valor
                    campo.dispatchEvent(new Event('change', { bubbles: true }));
                    campo.dispatchEvent(new Event('liveChange', { bubbles: true }));

                    // Busca o botão que acabou de ficar visível na tela
                    const botao = document.querySelector("[id$='nextButton-BDI-content']");
                    
                    if (botao) {
                        console.log("⏱️ 3 segundos atingidos! Apertando o botão Próximo...");
                        // Sequência completa de cliques físicos simulados para o SAP
                        botao.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, cancelable: true }));
                        botao.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, cancelable: true }));
                        botao.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true }));
                    } else {
                        console.error("❌ Erro: O botão apareceu, mas o ID mudou ou não pôde ser clicado.");
                    }
                }, 3000); 
            }
        });
    }

    // Criamos um observador que fica olhando a página inteira em tempo real
    const observador = new MutationObserver(function() {
        const segundoInput = document.querySelector("[id$='LoteInput2-inner']");
        
        // Se encontrar o segundo input na tela, ativa o monitoramento dele
        if (segundoInput) {
            iniciarMonitoramentoInput(segundoInput);
        }
    });

    // Configura o observador para vigiar toda a estrutura do site do SAP
    observador.observe(document.body, { childList: true, subtree: true });
})();
